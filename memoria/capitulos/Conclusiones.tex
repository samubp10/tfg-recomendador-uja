\section{Limitaciones}
\label{secc:limitaciones}

\textcolor{borrador2}{Este trabajo deja siete cosas sin cubrir. Cada una se
  argumenta en el capítulo anterior, junto al resultado que la provoca; aquí se
  enuncian juntas, con el apartado donde se razona, para que el lector no tenga
  que reconstruirlas.}

\begin{itemize}
  \item \textcolor{borrador2}{\textbf{El sistema no se ha probado con
      usuarios}, de modo que no se sabe si un estudiante formula sus dudas como
    las formula el banco de preguntas, ni si la respuesta le sirve para decidir
    (Apartado~\ref{secc:validacion_usuarios}).}

  \item \textcolor{borrador2}{\textbf{De cada respuesta se comprueba que no
      invente y que nombre lo que debía, no que sea buena}: una respuesta
    correcta, fundamentada y poco útil pasa todas las comprobaciones
    (Apartado~\ref{secc:metricas_no_capturan}).}

  \item \textcolor{borrador2}{\textbf{El banco de evaluación es propio y lo
      escribió la misma persona que escribió el corrector}
    (Apartado~\ref{secc:amenazas_validez}).}

  \item \textcolor{borrador2}{\textbf{Los tiempos de respuesta no son
      extrapolables a otro equipo}: se miden sobre el equipo doméstico que impone
    el requisito RNF-01, y ahí la mediana dobla el umbral fijado
    (Apartado~\ref{secc:amenazas_validez}).}

  \item \textcolor{borrador2}{\textbf{La colección cubre un solo centro y no
      todas sus titulaciones por igual}, de modo que el sistema sabe menos
    precisamente de las más nuevas (Apartado~\ref{secc:amenazas_validez}).}

  \item \textcolor{borrador2}{\textbf{La colección no recoge dos de las cuatro
      fuentes que enunciaba la propuesta}, las memorias de grado y los datos
    estadísticos (Apartado~\ref{secc:discusion_trazabilidad}).}

  \item \textcolor{borrador2}{\textbf{La aplicación web no está en condiciones
      de producción} (Apartado~\ref{secc:fase3_punto_entrada}).}
\end{itemize}

\section{Trabajo futuro}
\label{secc:trabajo_futuro}

\textcolor{borrador2}{Cada línea responde a una limitación de la lista
  anterior y en el mismo orden. Ninguna se enuncia si no se sigue de algo que
  este trabajo haya encontrado, y lo que haría falta para levantar cada una se
  razona en el Apartado~\ref{secc:levantar_amenazas}.}

\begin{itemize}
  \item \textcolor{borrador2}{\textbf{Probar el sistema con estudiantes de
      secundaria}, antes que cualquier otra cosa y antes de ponerlo en
    producción.}

  \item \textcolor{borrador2}{\textbf{Medir la utilidad de la respuesta y no
      solo su fidelidad}, que exige a su vez la validación con usuarios.}

  \item \textcolor{borrador2}{\textbf{Que el banco de preguntas y sus
      respuestas correctas los escriba alguien distinto de quien escribe el
    corrector.} Repetir las mediciones no levanta esa duda.}

  \item \textcolor{borrador2}{\textbf{Reducir el tiempo de respuesta} hasta el
    umbral que la hipótesis fijaba: la tarjeta gráfica para la inferencia, y
    menos llamadas al modelo por turno, que hoy son dos.}

  \item \textcolor{borrador2}{\textbf{Extender la colección al resto de centros
      de la Universidad.} El diseño admite crecer sin tocar el núcleo de
    recuperación y generación; que el sistema \emph{escale} \textbf{no está
      medido}, y es lo que habría que comprobar antes de prometerlo.}

  \item \textcolor{borrador2}{\textbf{Completar la colección con las memorias de
      grado y los datos estadísticos} que enunciaba la propuesta y que hoy no
    recoge.}

  \item \textcolor{borrador2}{\textbf{Llevar la aplicación a condiciones de
      producción}: atender en paralelo, cifrar el tráfico y limitar el tamaño y
    la frecuencia por cliente.}
\end{itemize}
