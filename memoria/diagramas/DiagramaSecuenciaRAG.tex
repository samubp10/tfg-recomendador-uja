% Diagrama de secuencia del recorrido completo de una pregunta (IT-41).
%
% Refleja la implementacion de IT-37, no un diseno previo: los tres desvios
% que se marcan son las tres salidas reales del codigo, y dos de ellas
% terminan SIN llamar al modelo, que es la decision de arquitectura que este
% diagrama existe para ensenar.
%
% La comprobacion de titulaciones NO tiene linea de vida propia y aparece como
% llamada del generador sobre si mismo, que es donde vive en el codigo. Darsela
% obligaba a ensanchar el dibujo una cuarta parte, y al ajustarlo al ancho de
% la pagina las etiquetas quedaban ilegibles.
\begin{figure}[H]
  \centering
  \resizebox{\textwidth}{!}{%
    \begin{tikzpicture}[
      lineaVida/.style={densely dotted, thick, draw=diagNodoBorde},
      mensaje/.style={-{Latex[length=2mm]}, semithick},
      devuelve/.style={-{Latex[length=2mm]}, semithick, densely dashed},
      etiqueta/.style={font=\small, midway, above, inner sep=1.5pt},
      propia/.style={font=\small, anchor=west, inner sep=1.5pt},
      desvio/.style={draw=diagGrupoBorde, semithick, densely dashed,
                     fill=diagGrupo, fill opacity=0.4, text opacity=1,
                     rounded corners=2pt},
      rotuloDesvio/.style={font=\small\bfseries, anchor=north west,
                           inner sep=2.5pt, text=diagGrupoBorde!65!black},
      apunte/.style={font=\small, align=center, draw=diagNodoBorde,
                     fill=white, inner sep=3pt, rounded corners=2pt},
    ]
      % --- Lineas de vida ---
      \foreach \x/\rotulo in {0/Estudiante, 3.4/Conversación, 7.2/Recuperador,
                              10.8/Generador, 14.4/Modelo local}{%
        \node[nodoDiag, minimum width=2.4cm, minimum height=0.8cm] at (\x,0) {\rotulo};
        \draw[lineaVida] (\x,-0.5) -- (\x,-15.2);
      }

      \draw[mensaje] (0,-1.1) -- node[etiqueta]{pregunta} (3.4,-1.1);
      \draw[mensaje] (3.4,-1.7) -- ++(0.8,0) -- ++(0,-0.45) -- ++(-0.8,0);
      \node[propia] at (4.3,-1.95) {resolver el ámbito y la anáfora};

      % --- Desvio 1: el mensaje no pide ningun dato ---
      \node[desvio, fit={(-1.3,-2.7) (15.7,-4.1)}] (d1) {};
      \node[rotuloDesvio] at (d1.north west) {el mensaje no pide un dato};
      \draw[devuelve] (3.4,-3.7) -- node[etiqueta]{respuesta fija} (0,-3.7);
      \node[apunte] at (10.4,-3.6)
        {un saludo, una despedida o preguntar por otro centro}
        [anchor=center];

      % --- Recuperacion ---
      \draw[mensaje] (3.4,-4.8) -- node[etiqueta]{consulta} (7.2,-4.8);
      \draw[mensaje] (7.2,-5.4) -- ++(0.8,0) -- ++(0,-0.45) -- ++(-0.8,0);
      \node[propia] at (8.1,-5.65) {incrustar y buscar los veinte más próximos};
      \draw[mensaje] (7.2,-6.4) -- ++(0.8,0) -- ++(0,-0.45) -- ++(-0.8,0);
      \node[propia] at (8.1,-6.65) {acotar: banda, corte relativo y suelo};
      \draw[devuelve] (7.2,-7.4) -- node[etiqueta]{fragmentos} (3.4,-7.4);

      % --- Desvio 2: no queda ningun fragmento ---
      \node[desvio, fit={(-1.3,-8.0) (15.7,-9.4)}] (d2) {};
      \node[rotuloDesvio] at (d2.north west) {no queda ningún fragmento};
      \draw[devuelve] (3.4,-9.0) -- node[etiqueta]{respuesta fija} (0,-9.0);
      \node[apunte] at (10.4,-8.9)
        {es el estado en el que un modelo inventa};

      % --- Generacion ---
      \draw[mensaje] (3.4,-10.1)
        -- node[etiqueta]{pregunta, fragmentos y turnos} (10.8,-10.1);
      \draw[mensaje] (10.8,-10.7) -- ++(0.8,0) -- ++(0,-0.45) -- ++(-0.8,0);
      \node[propia] at (11.7,-10.95) {armar el \textit{prompt}};
      \draw[mensaje] (10.8,-11.7) -- node[etiqueta]{\textit{prompt}} (14.4,-11.7);
      \draw[devuelve] (14.4,-12.3) -- node[etiqueta]{respuesta} (10.8,-12.3);
      \draw[mensaje] (10.8,-12.9) -- ++(0.8,0) -- ++(0,-0.45) -- ++(-0.8,0);
      \node[propia] at (11.7,-13.15)
        {comprobar cada titulación contra el catálogo};

      % --- Desvio 3: la barrera ---
      \node[desvio, fit={(-1.3,-13.8) (15.7,-15.2)}] (d3) {};
      \node[rotuloDesvio] at (d3.north west) {nombra alguna que no existe};
      \draw[devuelve] (10.8,-14.8)
        -- node[etiqueta]{respuesta retirada entera} (0,-14.8);

      % --- Salida por el camino feliz ---
      \draw[lineaVida] (0,-15.2) -- (0,-16.3);
      \draw[lineaVida] (10.8,-15.2) -- (10.8,-16.3);
      \draw[devuelve] (10.8,-15.9) -- node[etiqueta]{respuesta} (0,-15.9);
    \end{tikzpicture}%
  }
  \caption{\textcolor{borrador2}{Recorrido completo de una pregunta. Los tres
    bloques discontinuos son las salidas anticipadas del sistema; en las dos
    primeras la respuesta se entrega \textbf{sin llegar a consultar el
    modelo}.}}
  \label{fig:secuencia_rag}
\end{figure}
